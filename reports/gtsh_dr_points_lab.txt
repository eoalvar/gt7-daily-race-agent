function getProfile() {
        const np_online_id = document.getElementById('psnid').value;
        const getBtn = document.getElementById('getBtn'); // Gomb lekérése

        if (np_online_id.trim() === '') {
            alert('Please enter the Online ID!');
            return;
        }

        // Gomb letiltása a folyamat elején
        if (getBtn) {
            getBtn.disabled = true;
            getBtn.textContent = 'WAIT...'; // Opcionális: átírja a szöveget
        }

        const loader = document.getElementById('loader');
        if (loader) loader.classList.remove('hidden');

        const infoBox = document.getElementById('info-box');
        if (infoBox) infoBox.style.display = 'none';

        const drLimitBox = document.getElementById('info-box-dr-limit');
        if (drLimitBox) drLimitBox.style.display = 'none';

        document.getElementById('result').innerHTML = '';
        document.getElementById('saveImageBtn').style.display = 'none';

        fetch(window.location.href, {
            method: 'POST',
            headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
            body: 'psnid=' + encodeURIComponent(np_online_id)
        })
        .then(response => {
            if (!response.ok) {
                throw new Error('Network response was not ok: ' + response.statusText);
            }
            return response.json();
        })
        .then(responseData => {
            if (loader) loader.classList.add('hidden');

            if (responseData.error) {
                if (responseData.maintenance_notice) {
                    applyMaintenanceBox(responseData.maintenance_notice);
                }
                document.getElementById('result').innerHTML = `<p style="color: #ff3333;">Error: ${responseData.error}</p>`;
                document.getElementById('info-box').style.display = 'block';
                document.getElementById('info-box-error').style.display = 'block';
                return;
            }

            const key = document.body.getAttribute('header');
            if (!key) {
                console.error("Dinamikus kulcs nem található!");
                document.getElementById('result').innerHTML = `<p style="color: #ff3333;">Error: Missing encryption key</p>`;
                return;
            }

            try {
                const encryptedData = atob(responseData.data);
                const decryptedData = xorDecrypt(encryptedData, key);
                const data = JSON.parse(decryptedData);
                hideAutomaticMaintenanceBox();
                displayResults(data);
            } catch (error) {
                console.error("Dekódolási hiba:", error);
                document.getElementById('result').innerHTML = `<p style="color: #ff3333;">Error: Failed to decrypt data</p>`;
                document.getElementById('info-box').style.display = 'block';
                document.getElementById('info-box-error').style.display = 'block';
            }
        })
        .catch(error => {
            console.error('Fetch Error:', error);
            if (loader) loader.classList.add('hidden');
            document.getElementById('info-box').style.display = 'block';
            document.getElementById('info-box-error').style.display = 'block';
            document.getElementById('result').innerHTML = `<p style="color: #ff3333;">An error occurred: ${error.message}</p>`;
        })
        .finally(() => {
            // Ez a rész mindenképpen lefut a végén (siker és hiba esetén is)
            if (getBtn) {
                getBtn.disabled = false; // Gomb újra engedélyezése
                getBtn.textContent = 'GET'; // Szöveg visszaállítása
            }
        });
    }

FULL xorDecrypt() FUNCTION

function xorDecrypt(data, key) {
        let result = '';
        for (let i = 0; i < data.length; i++) {
            result += String.fromCharCode(data.charCodeAt(i) ^ key.charCodeAt(i % key.length));
        }
        return result;
    }
